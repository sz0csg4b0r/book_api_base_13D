Resource:
book/books {
    id: azonosito
    title: cím
    author: szerző
}

// JSON válasz
GET/book            - összes könyv lekérése
GET/book/:id        - egy könyv lekérése
POST/book           - egy könyv létrehozása
    {
        title: cím 
        author: szerző
    }
DELETE/book/:id     - egy könyv törlése
PATCH/book/:id      - egy könyv módosítása
    {
        title: cím 
        author: szerző
    }
POST/search         - könyv kérése
    {
        search: "bármi"
    }

//Böngésző
GET / - lista oldal : összeskönyv
GET /newbook - üres ürlap (szerző, könyvcím)
POST /newbook - új könyv mentése 
GET /editbook/:id  - kitöltött ürlap (szerző, könyvcím)
POST /editbook/:id - módosítás mentése
GET /deletebook/:id -törlése  

redirect : edit mentés után főoldalra
redirect : törlés után főoldalra
